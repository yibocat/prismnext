```ts
import type { CliParser } from "./types";
import type { ChatStreamMessage } from "@/stores/chat-store";

export class ClaudeParser implements CliParser {
  parse(line: string): ChatStreamMessage | null {
    let json: any;
    try {
      json = JSON.parse(line);
    } catch {
      return null;
    }

    // Skip system messages (session metadata)
    if (json.type === "system") return null;

    // assistant messages pass through directly
    if (json.type === "assistant") {
      return json as ChatStreamMessage;
    }

    // user messages (tool_result) pass through directly
    if (json.type === "user") {
      return json as ChatStreamMessage;
    }

    // result messages (completion)
    if (json.type === "result") {
      return {
        type: "result",
        usage: json.usage,
        duration_ms: json.duration_ms,
        result: json.subtype === "success"
          ? `Completed in ${json.duration_ms ?? 0}ms`
          : (json.error || "Unknown error"),
        is_error: !!(json.error),
      } as ChatStreamMessage;
    }

    return null;
  }
}
```

```ts
import { spawn, type ChildProcess } from "node:child_process";
import { BrowserWindow } from "electron";
import { getAgentConfig, DEFAULT_AGENT_ID } from "../agents/configs";
import { ClaudeParser } from "./claude-parser";
import type { CliSession, CliParser } from "./types";
import type { ChatStreamMessage } from "@/stores/chat-store";

const PARSERS: Record<string, CliParser> = {
  claude: new ClaudeParser(),
};

export class CliManager {
  private win: BrowserWindow;
  private sessions = new Map<string, CliSession>();
  private gatewayConfig: { baseUrl?: string; apiKey?: string } = {};

  constructor(win: BrowserWindow) {
    this.win = win;
  }

  setGateway(baseUrl?: string, apiKey?: string): void {
    this.gatewayConfig = { baseUrl, apiKey };
  }

  ensureSession(tabId: string, cwd: string, agentId: string = DEFAULT_AGENT_ID, model?: string | null): string {
    const existing = this.sessions.get(tabId);
    const sessionId = `cli-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;

    // Kill existing session for this tab if present
    if (existing) {
      try { existing.child.kill("SIGTERM"); } catch {}
      this.sessions.delete(tabId);
    }

    const config = getAgentConfig(agentId);
    if (!config) throw new Error(`Unknown agent: ${agentId}`);

    const args = [...config.args];
    if (model) args.push("--model", model);

    // If gateway is configured, set env vars for the child process
    const env: Record<string, string> = { ...process.env as Record<string, string> };
    if (this.gatewayConfig.baseUrl) {
      env.ANTHROPIC_BASE_URL = this.gatewayConfig.baseUrl;
      if (this.gatewayConfig.apiKey) env.ANTHROPIC_API_KEY = this.gatewayConfig.apiKey;
    }

    console.log(`[cli-manager] Spawning ${config.binary} ${args.join(" ")} for tab ${tabId}`);

    const child = spawn(config.binary, args, {
      cwd,
      env,
      stdio: ["pipe", "pipe", "pipe"],
    });

    const parser = PARSERS[agentId] || PARSERS.claude;

    let buffer = "";
    child.stdout!.on("data", (chunk: Buffer) => {
      buffer += chunk.toString();
      const lines = buffer.split("\n");
      buffer = lines.pop() || "";

      for (const line of lines) {
        const trimmed = line.trim();
        if (!trimmed) continue;
        const msg = parser.parse(trimmed);
        if (msg) {
          this.win.webContents.send("cli:stream", {
            tabId,
            data: JSON.stringify(msg),
          });
        }
      }
    });

    child.stderr!.on("data", (chunk: Buffer) => {
      this.win.webContents.send("cli:stderr", {
        tabId,
        data: chunk.toString(),
      });
    });

    child.on("exit", (code, signal) => {
      console.log(`[cli-manager] Process exited for tab ${tabId}: code=${code} signal=${signal}`);
      this.sessions.delete(tabId);
      this.win.webContents.send("cli:complete", {
        tabId,
        success: code === 0,
        error: code !== 0 ? `Process exited with code ${code}` : undefined,
      });
    });

    child.on("error", (err) => {
      console.error(`[cli-manager] Process error for tab ${tabId}:`, err.message);
      this.win.webContents.send("cli:complete", {
        tabId,
        success: false,
        error: err.message,
      });
    });

    const session: CliSession = {
      child,
      stdin: child.stdin!,
      sessionId,
      agentId,
      cwd,
      status: "idle",
      createdAt: Date.now(),
    };

    this.sessions.set(tabId, session);
    return sessionId;
  }

  sendPrompt(tabId: string, prompt: string): void {
    const session = this.sessions.get(tabId);
    if (!session) {
      this.win.webContents.send("cli:complete", {
        tabId,
        success: false,
        error: "No active session for this tab",
      });
      return;
    }

    session.status = "busy";
    const message = JSON.stringify({
      type: "user",
      message: { role: "user", content: prompt },
    });
    session.stdin.write(message + "\n");
  }

  cancel(tabId: string): void {
    const session = this.sessions.get(tabId);
    if (!session) return;
    console.log(`[cli-manager] Cancelling session for tab ${tabId}`);
    session.child.kill("SIGINT");
    // Session will be cleaned up in the "exit" handler
  }

  closeSession(tabId: string): void {
    const session = this.sessions.get(tabId);
    if (!session) return;
    console.log(`[cli-manager] Closing session for tab ${tabId}`);
    try { session.child.kill("SIGTERM"); } catch {}
    this.sessions.delete(tabId);
  }

  closeAll(): void {
    for (const [tabId, session] of this.sessions) {
      try { session.child.kill("SIGTERM"); } catch {}
    }
    this.sessions.clear();
  }

  getStatus(): { available: boolean; agentId?: string; agentName?: string; error?: string } {
    return { available: true, agentId: "claude", agentName: "Claude Code" };
  }
}
```

### Task 4: Create `main/ipc/cli.ts`

**Files:**
- Create: `src/main/ipc/cli.ts`

- [ ] **Step 1: Write IPC handlers**

```ts
import { ipcMain, BrowserWindow, app } from "electron";
import { CliManager } from "../cli/cli-manager";
import { listClaudeSessions, loadSessionHistory } from "../services/claude";
import type { ClaudeSession } from "../services/claude";

let cliManager: CliManager | null = null;

function getCliManager(win: BrowserWindow): CliManager {
  if (!cliManager) {
    cliManager = new CliManager(win);
  }
  return cliManager;
}

export function registerCliHandlers(): void {
  ipcMain.handle("cli:dispose", async () => {
    disposeCliManager();
    return { success: true };
  });

  ipcMain.handle("cli:prewarm", async (event, args: { projectPath: string; tabId?: string }) => {
    const tabId = args.tabId || "default";
    const win = BrowserWindow.fromWebContents(event.sender);
    if (!win) throw new Error("No window");
    const manager = getCliManager(win);
    const cwd = args.projectPath || app.getPath("home");
    manager.ensureSession(tabId, cwd);
    return { success: true };
  });

  ipcMain.handle("cli:status", async () => {
    const manager = cliManager;
    if (!manager) return { available: true, agentId: "claude", agentName: "Claude Code" };
    return manager.getStatus();
  });

  ipcMain.handle("cli:send", async (event, args: {
    projectPath: string; prompt: string; tabId?: string;
    agent?: string; model?: string | null;
  }) => {
    const tabId = args.tabId || "default";
    const win = BrowserWindow.fromWebContents(event.sender);
    if (!win) throw new Error("No window");
    const manager = getCliManager(win);
    const cwd = args.projectPath || app.getPath("home");
    manager.ensureSession(tabId, cwd, args.agent, args.model);
    manager.sendPrompt(tabId, args.prompt);
  });

  ipcMain.handle("cli:cancel", async (_event, args: { tabId?: string }) => {
    const manager = cliManager;
    if (!manager) return;
    manager.cancel(args.tabId || "default");
  });

  ipcMain.handle("cli:closeSession", async (_event, args: { tabId?: string }) => {
    const manager = cliManager;
    if (!manager) return;
    manager.closeSession(args.tabId || "default");
  });

  ipcMain.handle("cli:setGateway", async (_event, args: { baseUrl?: string; apiKey?: string }) => {
    const manager = cliManager;
    if (!manager) return;
    manager.setGateway(args.baseUrl, args.apiKey);
  });

  // Session management (still uses old Claude service for JSONL)
  ipcMain.handle("cli:listSessions", async (_event, args: { projectPath: string }) => {
    try {
      return await listClaudeSessions(args.projectPath || app.getPath("home"));
    } catch { return [] as ClaudeSession[]; }
  });

  ipcMain.handle("cli:loadSession", async (_event, args: { projectPath: string; sessionId: string }) => {
    try {
      return await loadSessionHistory(args.projectPath || app.getPath("home"), args.sessionId);
    } catch { return []; }
  });

  ipcMain.handle("cli:deleteSession", async (_event, args: { projectPath: string; sessionId: string }) => {
    try {
      const { unlink } = require("node:fs/promises");
      const { homedir } = require("node:os");
      const { join } = require("node:path");
      const encoded = args.projectPath?.replace(/[^a-zA-Z0-9]/g, "-") || "-";
      const sessionFile = join(homedir(), ".claude", "projects", encoded, `${args.sessionId}.jsonl`);
      await unlink(sessionFile);
      return { success: true };
    } catch (err: any) {
      return { success: false, error: err?.message || String(err) };
    }
  });
}

export function disposeCliManager(): void {
  if (cliManager) {
    cliManager.closeAll();
    cliManager = null;
  }
}
```
