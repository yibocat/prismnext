  \node[obs, below right=of z] (y) {\ylabel};
  \node[obs, below=1.35cm of z] (m) {\mlabel};
  \node[unobs, above=0.85cm of z] (u) {\ulabel};

  \draw[flow] (z) -- (y);
  \draw[flow] (x) -- (m);
