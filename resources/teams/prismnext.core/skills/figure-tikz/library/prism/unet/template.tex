\begin{tikzpicture}[

  node distance=\rowgap cm and \colgap cm,
  enc/.style={
    draw=otblue!75!black, fill=otblue!16, rounded corners=2pt,
    minimum width=1.7cm, minimum height=0.7cm,
    align=center, font=\sffamily\small,
  },
  bot/.style={
    draw=otorange!80!black, fill=otorange!20, rounded corners=2pt,

    align=center, font=\sffamily\small,
  },

  },

  \draw[flow] (input) -- (e1);
  \draw[flow] (e1) -- (e2);
  \draw[flow] (e2) -- (botn);
  \draw[flow] (botn) -- (d2);
  \draw[flow] (d2) -- (d1);
  \draw[flow] (d1) -- (output);
  \draw[skip] (e1) -- (d1);
  \draw[skip] (e2) -- (d2);
